// ============================================================
// Модуль формы внешней обработки «ВыгрузкаМетаданных»
//
// Что делает: выгружает метаданные активной конфигурации в JSON.
// Запуск: кнопка «Выгрузить в JSON» на форме.
// Сохранение: диалог выбора файла, по умолчанию config_export.json.
//
// Размещение: модуль формы внешней обработки.
// Все функции — &НаСервере, кроме самого обработчика кнопки.
//
// Версия 2 (2026-10-29):
//   - Диалог.ПредварительноеИмя → Диалог.ПолноеИмяФайла
//     (свойства ПредварительноеИмя в платформе нет)
//   - СобратьРеквизит больше не падает на измерениях/ресурсах
//     (у них нет свойства ЗаполнятьОбязательно)
// ============================================================

&НаКлиенте
Процедура ВыгрузитьМетаданные(Команда)
	
	Диалог = Новый ДиалогВыбораФайла(РежимДиалогаВыбораФайла.Сохранение);
	Диалог.Фильтр = "JSON (*.json)|*.json|Все файлы (*.*)|*.*";
	Диалог.Заголовок = "Сохранить выгрузку метаданных";
	Диалог.ПолноеИмяФайла = "config_export.json";
	Диалог.Расширение = "json";
	
	Если Диалог.Выбрать() Тогда
		ИмяФайла = Диалог.ПолноеИмяФайла;
		ВыгрузитьМетаданныеНаСервере(ИмяФайла);
		Сообщить("Готово! Файл сохранён: " + ИмяФайла);
	КонецЕсли;
	
КонецПроцедуры

&НаСервере
Процедура ВыгрузитьМетаданныеНаСервере(ИмяФайла)
	
	Результат = Новый Структура;
	
	// --- meta ---
	Мета = Новый Структура;
	Мета.Вставить("config_name", Метаданные.Имя);
	Мета.Вставить("config_synonym", Метаданные.Синоним);
	Мета.Вставить("config_version", Метаданные.Версия);
	Мета.Вставить("exported_at", XMLСтрока(ТекущаяДатаСеанса()));
	Результат.Вставить("meta", Мета);
	
	// --- documents ---
	Документы = Новый Массив;
	Для Каждого Док Из Метаданные.Документы Цикл
		Документы.Добавить(СобратьОбъект(Док, "Document", Истина));
	КонецЦикла;
	Результат.Вставить("documents", Документы);
	
	// --- catalogs ---
	Справочники = Новый Массив;
	Для Каждого Спр Из Метаданные.Справочники Цикл
		Справочники.Добавить(СобратьОбъект(Спр, "Catalog", Ложь));
	КонецЦикла;
	Результат.Вставить("catalogs", Справочники);
	
	// --- accumulation registers ---
	РегНак = Новый Массив;
	Для Каждого Рег Из Метаданные.РегистрыНакопления Цикл
		РегНак.Добавить(СобратьРегистр(Рег, "AccumulationRegister", Истина));
	КонецЦикла;
	Результат.Вставить("accumulation_registers", РегНак);
	
	// --- information registers ---
	РегСв = Новый Массив;
	Для Каждого Рег Из Метаданные.РегистрыСведений Цикл
		РегСв.Добавить(СобратьРегистр(Рег, "InformationRegister", Ложь));
	КонецЦикла;
	Результат.Вставить("information_registers", РегСв);
	
	// --- enumerations ---
	Перечисления = Новый Массив;
	Для Каждого Переч Из Метаданные.Перечисления Цикл
		Перечисления.Добавить(СобратьПеречисление(Переч));
	КонецЦикла;
	Результат.Вставить("enumerations", Перечисления);
	
	// --- serialize ---
	Запись = Новый ЗаписьJSON;
	Запись.УстановитьСтроку();
	ЗаписатьJSON(Запись, Результат);
	ТекстJSON = Запись.Закрыть();
	
	// --- save with UTF-8 BOM ---
	ЗаписьТекста = Новый ЗаписьТекста(ИмяФайла, КодировкаТекста.UTF8, Символы.ПС, Истина);
	ЗаписьТекста.Записать(ТекстJSON);
	ЗаписьТекста.Закрыть();
	
КонецПроцедуры

&НаСервере
Функция СобратьОбъект(ОбъектМД, ТипОбъекта, ЭтоДокумент)
	
	Рез = Новый Структура;
	Рез.Вставить("name", ОбъектМД.Имя);
	Рез.Вставить("synonym", ОбъектМД.Синоним);
	Рез.Вставить("type", ТипОбъекта);
	Рез.Вставить("comment", ОбъектМД.Комментарий);
	
	// Реквизиты
	Реквизиты = Новый Массив;
	Для Каждого Рекв Из ОбъектМД.Реквизиты Цикл
		Реквизиты.Добавить(СобратьРеквизит(Рекв));
	КонецЦикла;
	Рез.Вставить("attributes", Реквизиты);
	
	// Табличные части
	ТЧ = Новый Массив;
	Для Каждого ТабЧасть Из ОбъектМД.ТабличныеЧасти Цикл
		ТЧ.Добавить(СобратьТабличнуюЧасть(ТабЧасть));
	КонецЦикла;
	Рез.Вставить("tabular_sections", ТЧ);
	
	// Движения (только для документов)
	Если ЭтоДокумент Тогда
		Движения = Новый Массив;
		Для Каждого Движ Из ОбъектМД.Движения Цикл
			Движения.Добавить(Строка(Движ.Метаданные().ПолноеИмя()));
		КонецЦикла;
		Рез.Вставить("movements", Движения);
	КонецЕсли;
	
	Возврат Рез;
	
КонецФункции

&НаСервере
Функция СобратьРеквизит(Рекв)
	
	Рез = Новый Структура;
	Рез.Вставить("name", Рекв.Имя);
	Рез.Вставить("synonym", Рекв.Синоним);
	Рез.Вставить("type", Строка(Рекв.Тип));
	// ЗаполнятьОбязательно есть только у реквизитов, у измерений/ресурсов — нет
	Рез.Вставить("required", ?(Рекв.Свойство("ЗаполнятьОбязательно"), Рекв.ЗаполнятьОбязательно, Ложь));
	
	Возврат Рез;
	
КонецФункции

&НаСервере
Функция СобратьТабличнуюЧасть(ТабЧасть)
	
	Рез = Новый Структура;
	Рез.Вставить("name", ТабЧасть.Имя);
	Рез.Вставить("synonym", ТабЧасть.Синоним);
	
	Реквизиты = Новый Массив;
	Для Каждого Рекв Из ТабЧасть.Реквизиты Цикл
		Реквизиты.Добавить(СобратьРеквизит(Рекв));
	КонецЦикла;
	Рез.Вставить("attributes", Реквизиты);
	
	Возврат Рез;
	
КонецФункции

&НаСервере
Функция СобратьРегистр(Рег, ТипРегистра, ЭтоНакопления)
	
	Рез = Новый Структура;
	Рез.Вставить("name", Рег.Имя);
	Рез.Вставить("synonym", Рег.Синоним);
	Рез.Вставить("type", ТипРегистра);
	
	Если ЭтоНакопления Тогда
		Рез.Вставить("kind", Строка(Рег.ВидРегистра));
	Иначе
		Рез.Вставить("periodicity", Строка(Рег.Периодичность));
		Рез.Вставить("write_mode", Строка(Рег.РежимЗаписи));
	КонецЕсли;
	
	Измерения = Новый Массив;
	Для Каждого Изм Из Рег.Измерения Цикл
		Измерения.Добавить(СобратьРеквизит(Изм));
	КонецЦикла;
	Рез.Вставить("dimensions", Измерения);
	
	Ресурсы = Новый Массив;
	Для Каждого Рес Из Рег.Ресурсы Цикл
		Ресурсы.Добавить(СобратьРеквизит(Рес));
	КонецЦикла;
	Рез.Вставить("resources", Ресурсы);
	
	Реквизиты = Новый Массив;
	Для Каждого Рекв Из Рег.Реквизиты Цикл
		Реквизиты.Добавить(СобратьРеквизит(Рекв));
	КонецЦикла;
	Рез.Вставить("attributes", Реквизиты);
	
	Возврат Рез;
	
КонецФункции

&НаСервере
Функция СобратьПеречисление(Переч)
	
	Рез = Новый Структура;
	Рез.Вставить("name", Переч.Имя);
	Рез.Вставить("synonym", Переч.Синоним);
	Рез.Вставить("type", "Enum");
	
	Значения = Новый Массив;
	Для Каждого Знч Из Переч.ЗначенияПеречисления Цикл
		СтрЗнч = Новый Структура;
		СтрЗнч.Вставить("name", Знч.Имя);
		СтрЗнч.Вставить("synonym", Знч.Синоним);
		Значения.Добавить(СтрЗнч);
	КонецЦикла;
	Рез.Вставить("values", Значения);
	
	Возврат Рез;
	
КонецФ
